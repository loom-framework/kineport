window.addEventListener('DOMContentLoaded', function () {
    // Theme toggle logic
    const themeToggle = document.getElementById('theme-toggle');
    if (themeToggle) {
        themeToggle.addEventListener('click', function () {
            const currentTheme = document.documentElement.getAttribute('data-theme');
            const next = (currentTheme === 'dark') ? 'light' : 'dark';
            document.documentElement.setAttribute('data-theme', next);
            themeToggle.setAttribute('aria-pressed', next === 'dark' ? 'true' : 'false');
            try { localStorage.setItem('theme', next); } catch (e) {}
        });
        // On page load, set theme from localStorage if available
        try {
            const t = localStorage.getItem('theme');
            if (t) {
                document.documentElement.setAttribute('data-theme', t);
                themeToggle.setAttribute('aria-pressed', t === 'dark' ? 'true' : 'false');
            }
        } catch (e) {}
    }

    // Menu toggle logic
    const menuToggle = document.getElementById('menu-toggle');
    const navMenu = document.getElementById('nav-menu'); // Make sure you have an element with id="nav-menu"
    if (menuToggle && navMenu) {
        menuToggle.addEventListener('click', function () {
            const expanded = menuToggle.getAttribute('aria-expanded') === 'true';
            menuToggle.setAttribute('aria-expanded', !expanded ? 'true' : 'false');
            navMenu.style.display = expanded ? 'none' : 'block';
        });
    }
});
